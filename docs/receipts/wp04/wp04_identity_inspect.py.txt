def diff(x,y,path=''):

 if isinstance(x,dict):
  for k in x.keys()|y.keys():
   if k not in x or k not in y:print(path+'/'+k,'missing')
   elif x[k]!=y[k]:diff(x[k],y[k],path+'/'+k)
 elif isinstance(x,(list,tuple)):
  for i,(v,w) in enumerate(zip(x,y)):
   if v!=w:diff(v,w,path+'/'+str(i))
